"""Prepare, then explicitly execute a committed missing-slot v2 timing continuation."""

import argparse
import copy
from decimal import Decimal
import hashlib
import json
from pathlib import Path
import sys
import threading
import time

ROOT = Path(__file__).resolve().parents[2]
DIRECTORY = ROOT / "evaluations/decision-models"
sys.path.insert(0, str(DIRECTORY))
import run
from hosted_measurements import _budget, _request_cases, measure_hosted, panel_manifest
from measurements import _panel
from metrics import timing_metrics
from runners import DecisionRunner, SpendLedger
from supplement import build_supplement
from timing_continuation import _route, _summarize, execute_continuation, plan_continuation

CONFIG = ROOT / "evaluations/decision-models/configs/supplement-2026-10-28.json"
PARENT_LOCK = ROOT / "evaluations/decision-models/locks/supplement-timing-2026-10-28.json"
DEFAULT_REPORT = ROOT / "build/decision-model-evaluation/supplement-timing/report.json"


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def saved_runner(item, preflight):
    """Bind saved public metadata for estimation; no credential lookup or transport calls."""
    runner = DecisionRunner(item, environ={})
    runner._catalog = preflight["catalog"]
    runner._endpoints = preflight["endpoints"]
    runner._endpoint = next(row for row in preflight["endpoints"] if row["tag"] == item["upstream_tag"])
    runner._pricing = preflight["pricing"]
    runner._price = Decimal(str(preflight["pricing"]["prompt"]))
    return runner


def build_plan(report_path, config=None, corpus=None, original=None, parent=None):
    settings = run.read_json(CONFIG) if config is None else config
    corpus = build_supplement() if corpus is None else corpus
    original = run.read_json(report_path) if original is None else original
    parent = run.read_json(PARENT_LOCK) if parent is None else parent
    assert parent["config_sha256"] == run.digest(settings)
    assert parent["panel"] == panel_manifest(settings, corpus, []) == original["panel"]
    cases = {case["id"]: case for case in _panel(corpus, [])}
    models, total = [], Decimal(0)
    for item in settings["runners"]:
        if not item.get("enabled") or item.get("transport") != "openrouter":
            continue
        model = next(row for row in original["models"] if row["runner_id"] == item["id"])
        if model.get("record_count", 0) and "records" not in model:
            raise ValueError("Full timing receipts are required")
        metadata = model.get("preflight", {})
        if metadata.get("status") != "ok":
            raise ValueError("Saved route metadata missing: " + item["id"])
        partial = plan_continuation(original, [item], corpus, [])["models"][0]
        slots = {(row["case_id"], row["phase"], row.get("sample_index")) for row in partial["records"]}
        cold = [{"case_id": identifier, "indices": [index for index in range(5) if (identifier, "cold_client", index) not in slots]}
                for identifier in partial["cold_case_ids"]]
        warm = [{"case_id": cell["case_id"], "indices": cell["missing_indices"]} for cell in partial["cases"] if cell["missing_indices"]]
        cold = [cell for cell in cold if cell["indices"]]
        if not partial["records"]:
            mode = "fresh_standard_panel"
        elif cold:
            mode = "explicit_missing_cold_then_warm"
        elif warm:
            mode = "warm_continuation"
        else:
            mode = "complete_no_requests"
        estimator = saved_runner(item, metadata)
        bounds = {}
        for identifier in {cell["case_id"] for cell in [*cold, *warm]}:
            requests = _request_cases(cases[identifier], run.decision_rank)
            values = [estimator._payload(request, estimator._questions(request))[1] * estimator._price for request in requests]
            if any(value > Decimal(str(item.get("max_request_cost_usd", "0.10"))) for value in values):
                raise ValueError("A request exceeds the frozen cap")
            bounds[identifier] = sum(values, Decimal(0))
        bound = sum((bounds[cell["case_id"]] * len(cell["indices"]) for cell in cold), Decimal(0))
        bound += sum((bounds[cell["case_id"]] * (len(cell["indices"]) + 1) for cell in warm), Decimal(0))
        total += bound
        models.append({"runner_id": item["id"], "mode": mode, "original_record_count": len(partial["records"]),
                       "actual_atomic_attempts": sum(len(row.get("attempts", [])) for row in partial["records"]),
                       "remaining_cold": cold, "remaining_warm": warm,
                       "bound_usd": str(bound), "expected_route": _route(item, metadata),
                       "indexed_original_records_sha256": run.digest(partial["records"])})
    return {"experiment": "suggestions-v2-timing-missing-slot-continuation", "parent_lock_path": str(PARENT_LOCK.relative_to(ROOT)),
            "parent_lock_sha256": run.digest(parent), "original_report_path": str(Path(report_path).resolve().relative_to(ROOT)),
            "original_report_file_sha256": sha(report_path), "config_path": str(CONFIG.relative_to(ROOT)),
            "config_sha256": run.digest(settings), "corpus_sha256": run.digest(corpus), "panel_hash": original["panel"]["panel_hash"],
            "source_fingerprint": run.source_fingerprint(), "helper_file_sha256": sha(__file__), "hardware": run.hardware(),
            "latency_concurrency": 1, "total_remaining_bound_usd": str(total), "models": models,
            "pacing_after_record_ms": {"openrouter-clef-flash": 1100},
            "policies": ["Completed and failed indexed slots remain consumed; no automatic retries or replacements.",
                         "Exactly the listed missing cold indices may use fresh clients; this never establishes server/model cold state.",
                         "One excluded warmup is added per incomplete warm cell; labels, routes, prompts and thresholds remain frozen.",
                         "Clef receives 1100ms declared idle after each new receipt checkpoint, outside request timers; preserve original-unpaced versus paced strata.",
                         "Zero-record models use the standard frozen panel; all known source outcomes remain in final artifacts.",
                         "The entire remaining bound must fit the reconciled shared experiment ledger before any POST."]}


def execute(plan, output):
    settings, corpus = run.read_json(CONFIG), build_supplement()
    original = run.read_json(ROOT / plan["original_report_path"])
    expected = build_plan(ROOT / plan["original_report_path"], settings, corpus, original)
    assert expected == plan and sha(__file__) == plan["helper_file_sha256"]
    assert not output.exists(), "Preserve prior continuation outputs"
    ledger_path = ROOT / settings["ledger_path"]
    canonical = ROOT / "build/decision-model-evaluation/spend-ledger.json"
    assert ledger_path.resolve() == canonical.resolve()
    output.mkdir(parents=True)
    cancelled = threading.Event()
    def save_ledger(state):
        temporary = ledger_path.with_suffix(".tmp")
        run.save_json(temporary, state)
        temporary.replace(ledger_path)
    with run.experiment_lock(ledger_path):
        ledger = SpendLedger.from_dict(run.read_json(ledger_path), checkpoint=save_ledger)
        remaining = _budget(ledger)
        assert remaining is not None and Decimal(plan["total_remaining_bound_usd"]) <= remaining
        clients = {}
        for model in plan["models"]:
            if model["mode"] == "complete_no_requests":
                continue
            item = next(row for row in settings["runners"] if row["id"] == model["runner_id"])
            client = DecisionRunner(item, ledger=ledger, cancel_event=cancelled)
            status = client.preflight()
            assert status.get("status") == "ok" and _route(item, status) == model["expected_route"]
            clients[model["runner_id"]] = client
        merged = copy.deepcopy(original)
        cases = {case["id"]: case for case in _panel(corpus, [])}
        with (output / "new-receipts.jsonl").open("x", encoding="utf-8") as stream:
            def capture(row):
                idle = plan["pacing_after_record_ms"].get(row["runner_id"], 0)
                row.update(quality_evaluation=False, idle_after_ms=idle,
                           idle_after_ms_semantics="Declared pause after this checkpoint; excluded from request latency.",
                           pacing_stratum="clef_paced_1100ms_after_record" if idle else "continuation_unpaced")
                stream.write(json.dumps(row, ensure_ascii=False) + "\n")
                stream.flush()
                if idle and not cancelled.is_set():
                    time.sleep(idle / 1000)
            for model in plan["models"]:
                if cancelled.is_set() or _budget(ledger) is None:
                    break
                if model["mode"] == "complete_no_requests":
                    continue
                item = next(row for row in settings["runners"] if row["id"] == model["runner_id"])
                prior = next(row for row in merged["models"] if row["runner_id"] == item["id"])
                if model["mode"] == "fresh_standard_panel":
                    def checked_factory(selected, ledger, cancel_event=None):
                        client = DecisionRunner(selected, ledger=ledger, cancel_event=cancel_event)
                        status = client.preflight()
                        assert status.get("status") == "ok" and _route(selected, status) == model["expected_route"]
                        return client
                    report = measure_hosted([item], corpus, [], ledger, runner_factory=checked_factory, cancel_event=cancelled, checkpoint=capture)
                    report["models"][0]["measurement"] = "measured_openrouter_end_to_end"
                    report["models"][0]["measurement_factory"] = "Live DecisionRunner with operational frozen-route validation"
                    prior.update(report["models"][0])
                else:
                    indexed = plan_continuation(merged, [item], corpus, [])["models"][0]
                    prior["records"] = indexed["records"]
                    for cell in model["remaining_cold"]:
                        case = cases[cell["case_id"]]
                        for index in cell["indices"]:
                            if cancelled.is_set() or _budget(ledger) is None:
                                break
                            fresh = DecisionRunner(item, ledger=ledger, cancel_event=cancelled)
                            status = fresh.preflight()
                            assert status.get("status") == "ok" and _route(item, status) == model["expected_route"]
                            result = fresh.predict(case)
                            record = {**result, "runner_id": item["id"], "case_id": case["id"], "task": case["task"],
                                      "language": case["language"], "phase": "cold_client", "sample_index": index,
                                      "planned_sample_index": index + 1, "quality_evaluation": False}
                            prior["records"].append(record)
                            capture(record)
                    prior["record_count"] = len(prior["records"])
                    if _budget(ledger) is not None and not cancelled.is_set():
                        report = execute_continuation(merged, [item], corpus, [], ledger, cancel_event=cancelled, checkpoint=capture)
                        continued = report["models"][0]
                        prior.update(continued)
                prior["record_count"] = len(prior.get("records", []))
                run.save_json(output / "report.json", {**merged, "continuation_plan_sha256": run.digest(plan), "ledger": ledger.snapshot()})
        indexed = plan_continuation(merged, settings, corpus, [])
        for model in indexed["models"]:
            _summarize(model)
            target = next(row for row in merged["models"] if row["runner_id"] == model["runner_id"])
            target.update(model)
            strata = {}
            for record in model["records"]:
                if record["phase"] == "warm":
                    key = (record.get("pacing_stratum", "original_unpaced"), record["language"], record["task"])
                    strata.setdefault(key, []).append(record)
            target["pacing_strata"] = [{"stratum": key[0], "language": key[1], "task": key[2], "consumed_slots": len(rows),
                                       "status_counts": {status: sum(row["status"] == status for row in rows) for status in sorted({row["status"] for row in rows})},
                                       "successful_request_latency": timing_metrics([row["latency_ms"] for row in rows if row["status"] == "ok" and isinstance(row.get("latency_ms"), (int, float))])}
                                      for key, rows in sorted(strata.items())]
        complete = all(row.get("complete_coverage") for row in merged["models"])
        passed = complete and all(cell["all_passed"] for row in merged["models"] for cell in row["cases"])
        run.save_json(output / "report.json", {**merged, "continuation_plan_sha256": run.digest(plan), "ledger": ledger.snapshot(),
                                               "complete_coverage": complete, "all_passed": passed,
                                               "pacing_after_record_ms": plan["pacing_after_record_ms"],
                                               "pacing_scope": "Declared idle occurs after receipt checkpoint and outside request timing; strata preserve changed operational pacing.",
                                               "status": "complete_coverage" if complete else "partial"})


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--frozen-path", "--frozenPath", type=Path)
    parser.add_argument("--output-path", "--outputPath", type=Path, required=True)
    parser.add_argument("--report", type=Path, default=DEFAULT_REPORT)
    args = parser.parse_args()
    if args.prepare:
        if not args.report.is_file():
            parser.error("Original timing report is not finished; no continuation plan created")
        assert not args.output_path.exists(), "Preserve earlier plans"
        plan = build_plan(args.report)
        run.save_json(args.output_path, plan)
        print(json.dumps({"status": "prepared_no_requests", "plan_sha256": run.digest(plan), "bound_usd": plan["total_remaining_bound_usd"],
                          "models": [{"runner_id": row["runner_id"], "mode": row["mode"]} for row in plan["models"]]}))
        return
    if args.frozen_path is None:
        parser.error("--execute requires --frozen-path containing the reviewed, committed continuation plan")
    relative = str(args.frozen_path.resolve().relative_to(ROOT))
    assert run.git("ls-files", "--", relative) and not run.git("status", "--porcelain", "--", relative, "evaluations/decision-models", "core/src")
    execute(run.read_json(args.frozen_path), args.output_path)


if __name__ == "__main__":
    main()
