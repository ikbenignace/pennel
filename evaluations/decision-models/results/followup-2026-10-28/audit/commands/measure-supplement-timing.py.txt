import json,sys
from pathlib import Path
sys.path.insert(0,'evaluations/decision-models')
import run
from supplement import build_supplement
from hosted_measurements import panel_manifest,measure_hosted
from runners import SpendLedger
root=Path('build/decision-model-evaluation');output=root/'supplement-timing'
config=run.read_json('evaluations/decision-models/configs/supplement-2026-10-28.json')
freeze=run.read_json('evaluations/decision-models/locks/supplement-timing-2026-10-28.json')
corpus=build_supplement()
assert freeze['parent_manifest_sha256']==run.digest(run.read_json('evaluations/decision-models/locks/supplement-2026-10-28.json'))
assert freeze['config_sha256']==run.digest(config)
assert freeze['panel']==panel_manifest(config,corpus,[])
assert freeze['hardware']==run.hardware()
assert not run.git('status','--porcelain','--','evaluations/decision-models','core/src')
assert not output.exists()
output.mkdir()
ledger_path=Path(config['ledger_path'])
def checkpoint(state):
 tmp=ledger_path.with_suffix('.tmp');run.save_json(tmp,state);tmp.replace(ledger_path)
with run.experiment_lock(ledger_path):
 ledger=SpendLedger.from_dict(run.read_json(ledger_path),checkpoint=checkpoint)
 assert not ledger.unresolved and not ledger.pending
 count=0
 with (output/'receipts.jsonl').open('x',encoding='utf-8') as stream:
  def capture(row):
   global count
   stream.write(json.dumps(row,ensure_ascii=False)+'\n');stream.flush();count+=1
   if count%100==0 or row.get('status')!='ok':print(json.dumps({'records':count,'runner_id':row['runner_id'],'status':row['status'],'reason':row.get('reason'),'total_billed_usd':str(ledger.spent)}),flush=True)
  report=measure_hosted(config,corpus,[],ledger,checkpoint=capture)
 report['run_manifest_hash']=run.digest(freeze);report['generation_commit']=run.git('rev-parse','HEAD');report['client_hardware']=run.hardware()
 run.save_json(output/'report.json',report)
 print(json.dumps({'status':report['status'],'records':count,'total_billed_usd':str(ledger.spent),'unresolved':ledger.unresolved}),flush=True)
