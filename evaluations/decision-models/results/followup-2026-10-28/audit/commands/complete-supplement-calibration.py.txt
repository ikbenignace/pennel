import json,sys
from pathlib import Path
sys.path.insert(0,'evaluations/decision-models')
import run,followup
from supplement import build_supplement
root=Path('build/decision-model-evaluation')
corpus=build_supplement();config=run.read_json('evaluations/decision-models/configs/supplement-2026-10-28.json');frozen=run.read_json('evaluations/decision-models/locks/supplement-2026-10-28.json')
rows=[json.loads(line) for line in (root/'supplement-calibration/predictions.jsonl').read_text().splitlines()]
unsent=[r for r in rows if not r.get('attempts') and r.get('status')=='unavailable'];assert len(unsent)==1
keys={(r['runner_id'],r['case_id']) for r in unsent};cases={c['id']:c for c in corpus['decisions']}
assert len({r['runner_id'] for r in unsent})==1
runner_id=unsent[0]['runner_id'];newconfig={**config,'runners':[r for r in config['runners'] if r['id']==runner_id]}
subset={**corpus,'decisions':[cases[r['case_id']] for r in unsent]}
new,available=run.live_predictions(subset,{'queries':[],'passages':[]},newconfig,'calibration',root/'supplement-calibration-one-unsent',frozen)
assert {(r['runner_id'],r['case_id']) for r in new}==keys
index={(r['runner_id'],r['case_id']):r for r in new};merged=[index.get((r['runner_id'],r['case_id']),r) for r in rows]
assert len(merged)==1080 and sum(bool(r.get('attempts')) for r in merged)==1080
followup.validate_predictions(merged,frozen,corpus,'calibration')
artifact=followup.fit_recorded(merged,frozen,corpus)
output=root/'supplement-calibration-complete';output.mkdir(exist_ok=False)
run.save_json(output/'manifest.json',frozen);run.save_json(output/'predictions.json',{'frozen':run.digest(frozen),'rows':merged});run.save_json(output/'calibration.json',artifact)
(output/'predictions.jsonl').write_text(''.join(json.dumps(r,ensure_ascii=False)+'\n' for r in merged))
run.save_json(output/'availability.json',run.read_json(root/'supplement-calibration/availability.json'))
run.save_json(output/'execution.json',{'split':'calibration','status':'complete','run_manifest_hash':run.digest(frozen),'calibration_sha256':run.digest(artifact),'predictions_sha256':run.digest(merged),'continuation':{'only_previously_unsent_cases':list(keys),'failed_and_completed_paid_cases_preserved':True}})
run.save_json(output/'report.json',followup._report(merged,artifact,frozen,corpus,'calibration',available))
print(json.dumps({'rows':len(merged),'status_counts':dict(__import__('collections').Counter(r['status'] for r in merged)),'calibration_sha256':run.digest(artifact)}))
